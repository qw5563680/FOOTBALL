在有固定数据提交的仓库中运行下列命令。仅提取必需的六个文件，不恢复或更新正式数据库。

```bash
t100_archive=$(mktemp -d)
t100_pin=f87ae9232653add8d5da2f23138a028ca6896697
mkdir -p "$t100_archive/state/audit"
for path in feed_manifest.json state/{teams,matches,results}.jsonl \
  state/audit/{apif,fdorg}_identity_merges.csv; do
  git show "$t100_pin:$path" > "$t100_archive/$path" || exit 1
done
python examples/t100_identity_audit.py --archive "$t100_archive"
```
